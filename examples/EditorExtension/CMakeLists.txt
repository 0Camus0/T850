cmake_minimum_required(VERSION 3.21)
cmake_policy(SET CMP0091 NEW)
project(EditorExtension LANGUAGES C CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(T850_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../T850" CACHE PATH "T850 source root")
set(T850_BUILD_EDITOR ON CACHE BOOL "Build the editor SDK" FORCE)
add_subdirectory("${T850_ROOT}" t850)
add_executable(EditorExtension main.cpp)
target_link_libraries(EditorExtension PRIVATE T850::Editor)
t850_stage_editor_runtime(EditorExtension ENGINE_ASSETS)
